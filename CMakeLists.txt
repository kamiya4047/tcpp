cmake_minimum_required(VERSION 3.24)
project(TaiwanBopomofo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(IME_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
option(IME_BUILD_WINDOWS_TSF "Build the native Windows TSF adapter" ${WIN32})
option(IME_ENABLE_DICTIONARY "Bundle the canonical indexed dictionary" ON)
include(CTest)
add_library(ime_core STATIC src/core/types.cpp src/core/engine.cpp src/core/settings.cpp src/core/user_dictionary.cpp src/bopomofo/bopomofo.cpp src/providers/providers.cpp)
target_include_directories(ime_core PUBLIC include)
target_compile_definitions(ime_core PRIVATE IME_ENABLE_DICTIONARY=$<BOOL:${IME_ENABLE_DICTIONARY}>)
function(ime_copy_dictionary target)
  if(IME_ENABLE_DICTIONARY AND EXISTS "${CMAKE_SOURCE_DIR}/data/lexicon/dictionary.idx")
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/data/lexicon"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/data/lexicon/dictionary.idx" "$<TARGET_FILE_DIR:${target}>/data/lexicon/dictionary.idx")
  endif()
endfunction()
if(MSVC)
  target_compile_options(ime_core PUBLIC /utf-8 /W4 $<$<BOOL:${IME_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(ime_core PUBLIC -Wall -Wextra -Wpedantic $<$<BOOL:${IME_WARNINGS_AS_ERRORS}>:-Werror>)
endif()
add_executable(decoder_cli src/devtools/decoder_cli.cpp)
target_link_libraries(decoder_cli PRIVATE ime_core)
ime_copy_dictionary(decoder_cli)
if(MINGW)
  target_link_options(decoder_cli PRIVATE -municode)
endif()
add_executable(ime_tests tests/test_main.cpp tests/decoder_tests.cpp tests/provider_tests.cpp tests/qa_tests.cpp tests/user_dictionary_tests.cpp)
target_link_libraries(ime_tests PRIVATE ime_core)
ime_copy_dictionary(ime_tests)
add_test(NAME ime_tests COMMAND ime_tests)
if(IME_BUILD_WINDOWS_TSF)
  if(NOT WIN32)
    message(FATAL_ERROR "IME_BUILD_WINDOWS_TSF requires Windows")
  endif()
  add_subdirectory(src/windows_tsf)
endif()
install(TARGETS decoder_cli RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION . OPTIONAL)
install(FILES tools/register_ime.ps1 DESTINATION tools)
install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md")
install(FILES data/SOURCES.md DESTINATION data)
if(IME_ENABLE_DICTIONARY AND EXISTS "${CMAKE_SOURCE_DIR}/data/lexicon/dictionary.idx")
  install(FILES data/lexicon/dictionary.idx DESTINATION bin/data/lexicon)
endif()
set(CPACK_PACKAGE_NAME "TaiwanBopomofo")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR "ZIP")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
include(CPack)
